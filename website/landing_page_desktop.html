import lens_core
from security import AirGapProtocol

# Initialize secure enclave
engine = lens_core.Engine(
    mode="local_strict",
    protocol=AirGapProtocol.V2
)

# Analyze incoming payload
def inspect_prompt(payload):
    metrics = engine.evaluate(payload)
    if metrics.risk_score > 0.8:
        return engine.redact(payload)
    return metrics.approved_stream()
